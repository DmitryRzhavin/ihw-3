cmake_minimum_required(VERSION 3.16)
project(restaurant_processes C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
foreach(pair "restaurant_unnamed;0" "restaurant_named;1" "restaurant_shm;2" "restaurant_mq;3")
    list(GET pair 0 target)
    list(GET pair 1 backend)
    add_executable(${target} src/main.c src/model.c src/config.c)
    target_include_directories(${target} PRIVATE include)
    target_compile_definitions(${target} PRIVATE _GNU_SOURCE BACKEND=${backend})
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(${target} PRIVATE pthread rt)
    if(ENABLE_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endforeach()
enable_testing()
add_test(NAME unnamed COMMAND restaurant_unnamed --config ${CMAKE_SOURCE_DIR}/configs/demo.conf --unit-ms 1 --no-log)
add_test(NAME named COMMAND restaurant_named --config ${CMAKE_SOURCE_DIR}/configs/demo.conf --unit-ms 1 --no-log)
set_tests_properties(unnamed named PROPERTIES TIMEOUT 15)
add_executable(model_test tests/model_test.c src/model.c)
target_include_directories(model_test PRIVATE include)
target_compile_definitions(model_test PRIVATE _GNU_SOURCE)
target_compile_options(model_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME model_invariants COMMAND model_test)
if(ENABLE_SANITIZERS)
    target_compile_options(model_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(model_test PRIVATE -fsanitize=address,undefined)
endif()
